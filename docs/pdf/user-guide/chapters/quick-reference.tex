\section{Redis CLI — Common Operations}
\index{Redis!CLI commands}

\begin{table}[ht]
  \small
  \begin{tabularx}{\linewidth}{lX}
    \toprule
    \textbf{Command} & \textbf{Effect} \\
    \midrule
    \code{redis-cli PING} & Connectivity check \\
    \code{redis-cli GET "ban:1.2.3.4"} & Check if IP is banned \\
    \code{redis-cli DEL "ban:1.2.3.4"} & Clear a specific ban \\
    \code{redis-cli SETEX "ban:1.2.3.4" 300 "manual"} & Ban IP for 5 minutes \\
    \code{redis-cli --scan --pattern "ban:*" | wc -l} & Count active bans \\
    \code{redis-cli SADD ja4:blacklist "fingerprint"} & Add to JA4 blacklist \\
    \code{redis-cli SREM ja4:blacklist "fingerprint"} & Remove from blacklist \\
    \code{redis-cli SISMEMBER ja4:blacklist "fp"} & Check if fingerprint is blacklisted \\
    \code{redis-cli GET "config:dial"} & Read current dial value \\
    \code{redis-cli SET "config:dial" "50"} & Set dial to 50 (affects new connections) \\
    \code{redis-cli FLUSHALL} & \textbf{DEV ONLY} — wipe all Redis data \\
    \bottomrule
  \end{tabularx}
\end{table}

Prefix all commands with \code{docker compose exec redis} when running from the host:
\begin{bashcode}
docker compose exec redis redis-cli DEL "ban:1.2.3.4"
\end{bashcode}

\section{Config Reload}
\index{config reload!quick reference}

\begin{bashcode}
# Hot reload (no restart required)
kill -HUP $(docker compose exec ja4proxy cat /var/run/proxy.pid)

# Verify reload succeeded
docker compose logs ja4proxy | tail -3
# Expected: INFO | config | event=reload | status=ok
\end{bashcode}

\section{Scaling}
\index{scaling!quick reference}

\begin{bashcode}
# Docker Compose scaling
./scripts/scale-proxies.sh 4   # Scale to 4 proxy instances

# Kubernetes
helm install ja4proxy ./deploy/helm/ja4proxy \
  --set proxy.backendHost=your.backend \
  --namespace ja4proxy --create-namespace

kubectl get hpa -n ja4proxy    # Check autoscaler status
kubectl rollout restart deployment/ja4proxy -n ja4proxy  # Rolling restart
\end{bashcode}
